"""Gate all 600 successful captures without reading reference labels or ranks."""
from pathlib import Path
from collections import Counter
import hashlib,json
OUT=Path(__file__).resolve().parent; BASE=OUT.parents[2]
def sha(p):
 with Path(p).open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def load(p):return json.loads(Path(p).read_text())
def preflight():
 workers=0; calls=0; artifacts=0
 for root,arms in [(BASE/'output/analysis/embedding-language-coverage-v320',['baseline','jina']),(OUT,['muninn'])]:
  reg=load(root/'registration.json'); closure=load(root/'capture/closure.json')
  assert closure['all_outputs_closed'] and closure['registration_sha256']==sha(root/'registration.json')
  assert sha(root/'capture.py')==reg['producer_sha256'] and sha(root/'prefill.py')==reg['prefill_sha256']
  assert not list((root/'capture').rglob('failure.json'))
  for name,digest in closure['pins'].items():assert sha(root/'capture'/name)==digest;artifacts+=1
  requests=load(root/'requests.json'); sources=load(root/'sources.json')
  assert len(requests)==200 and len({r['id'] for r in requests})==200
  assert sha(root/'requests.json')==reg['requests_sha256'] and sha(root/'sources.json')==reg['sources_sha256']
  for group in sources:
   expected=Counter(r['id'] for r in requests if r['repo_id']==group); assert len(expected)==10
   for arm in arms:
    folder=root/'capture'/group/arm; prov=load(folder/'provenance.json')
    assert prov['complete'] and prov['heavy_imports']==[] and prov['provider']==['CPUExecutionProvider'] and prov['reranker_tokens']==1024
    responses=load(folder/'responses.json'); raw=load(folder/'raw.json')
    assert Counter(r['id'] for r in responses)==Counter(r['id'] for r in raw)==expected
    assert all('error' not in r and r.get('ok') is not False and len(r['results'])<=10 for r in responses)
    for name in ['responses.json','raw.json','provenance.json','owners.json','cache.json']:
     assert str((folder/name).relative_to(root/'capture')) in closure['pins']
    workers+=1; calls+=len(responses)
 assert workers==60 and calls==600
 result={'successful_workers':workers,'successful_unique_calls':calls,'pinned_artifacts':artifacts,'reference_labels_read':False,'reserve':'CLOSED'}
 (OUT/'preflight.json').write_text(json.dumps(result,indent=2)+'\n')
 return result
if __name__=='__main__':print(json.dumps(preflight()))
