"""Post-capture paired diagnostics; no fitting or model selection loop."""
from pathlib import Path
import hashlib,json
import numpy as np
OUT=Path(__file__).resolve().parent;BASE=OUT.parents[2];CONTROL=BASE/'output/analysis/embedding-language-coverage-v320'
assert (OUT/'finalized.json').exists()
report=json.loads((OUT/'report.json').read_text());rows=report['rows']
groups=sorted({r['repo_id'] for r in rows});assert len(groups)==20
arms=['baseline','jina','muninn'];metrics=['top1','top10']
by_repo={g:{a:{m:sum(r[m] for r in rows if r['repo_id']==g and r['arm']==a and r['stage']=='final') for m in metrics} for a in arms} for g in groups}
rng=np.random.default_rng(20261017)
strata=[[i for i,g in enumerate(groups) if g.startswith(language+'--')] for language in ['cpp','rust']]
assert list(map(len,strata))==[10,10]
samples=np.concatenate([rng.choice(indices,size=(10000,10),replace=True) for indices in strata],axis=1);paired={}
for arm in arms[1:]:
 paired[arm]={}
 for metric in metrics:
  differences=np.array([by_repo[g][arm][metric]-by_repo[g]['baseline'][metric] for g in groups],dtype=float)
  estimates=differences[samples].sum(axis=1)/200
  paired[arm][metric]={'net_queries':int(differences.sum()),'net_percentage_points':float(differences.sum()/2),'repository_bootstrap_95_interval_percentage_points':(100*np.quantile(estimates,[.025,.975])).tolist(),'bootstrap_replicates':10000,'bootstrap_unit':'repository, paired and stratified by language (10CPP+10Rust)' ,'seed':20261017}
prefill={}
for arm in arms:
 root=OUT if arm=='muninn' else CONTROL
 values=[]
 for group in groups:
  folder=root/'capture'/group/arm
  values.append({'repo_id':group,**json.loads((folder/'prefill.json').read_text())})
 prefill[arm]=values
summary={'per_repository':by_repo,'paired':paired,'source_preparation':prefill,'interpretation':'Descriptive paired differences on200 exposed public queries nested in20 repos. Bootstrap resamples paired repositories within each language, preserving 100CPP/100Rust; not individual queries. No blind acceptance claim. Source prep cache state is mixed for control/Jina and cold for Muninn vectors; do not compare as fair cold-index costs.','reserve':'CLOSED','report_sha256':hashlib.sha256((OUT/'report.json').read_bytes()).hexdigest()}
(OUT/'additional-analysis.json').write_text(json.dumps(summary,indent=2)+'\n');print(json.dumps({'paired':paired,'per_repository':by_repo},indent=2))
