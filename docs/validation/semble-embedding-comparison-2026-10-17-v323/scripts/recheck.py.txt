"""Recheck archived locations, metrics and costs without models or source trees."""
import argparse, gzip, hashlib, json, math, runpy, sys
from pathlib import Path
import numpy as np

p=argparse.ArgumentParser()
p.add_argument('archive',type=Path)
p.add_argument('--tool-root',type=Path,required=True)
args=p.parse_args();root=args.archive.resolve()
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
inventory=json.loads((root/'archive-inventory.json').read_text())
for n,d in inventory['files'].items():assert sha(root/n)==d,n
reg=json.loads((root/'registration.json').read_text())
for n,d in reg['scorer_source_pins'].items():
    target=args.tool_root/'benchmarks'/n.split('/benchmarks/',1)[1]
    assert sha(target)==d,n
for n,d in reg['producer_pins'].items():
    target=root/'scripts'/(n+'.txt') if n.endswith('.py') else root/n
    assert sha(target)==d,n
assert sha(root/'engine-source-manifest.json')==reg['engine_manifest_sha256']
assert sha(root/'sources.json')==reg['sources_sha256']
assert sha(root/'requests.jsonl')==reg['requests_sha256']
sys.path.insert(0,str(args.tool_root))
evaluate=runpy.run_path(str(root/'scripts/original_adapter.py.txt'),run_name='archived_adapter')['evaluate_rows']
from benchmarks.data import load_tasks,load_repo_specs
tasks=load_tasks(load_repo_specs());assert len(tasks)==1251
summary=json.loads((root/'summary.json').read_text())
by_arm={}
for filename in ['query-observations.jsonl.gz','historical-quality-observations.jsonl.gz']:
    with gzip.open(root/filename,'rt') as f:
        for line in f:
            r=json.loads(line);rows=by_arm.setdefault(r['arm'],{})
            assert r['id'] not in rows;rows[r['id']]=r
for arm,rows in by_arm.items():
    assert len(rows)==1251
    r=evaluate(tasks,rows)
    if arm=='ken-8ce6b00-historical-v219':
        expected=json.loads((root/'audit.json').read_text())['historical_ken']['metrics']
    else:expected=summary['metrics'][arm]
    r['top1_count']=sum(q['top1'] for q in r['per_query'])
    r['top10_count']=sum(q['hit10'] for q in r['per_query'])
    for k,v in expected.items():assert r[k]==v,(arm,k,r[k],v)
provenance=json.loads((root/'worker-provenance.json').read_text())
preparations=json.loads((root/'preparation-observations.json').read_text())
def dist(xs):return {k:float(np.percentile(xs,v)) for k,v in [('p50',50),('p95',95),('max',100)]}
for arm in reg['arms']:
    rows=list(by_arm[arm].values());provs=[p for p in provenance if p['arm']==arm]
    setups=[p['setup'] for p in preparations];prefills=[p['arms'][arm] for p in preparations]
    assert len(rows)==1251 and len(provs)==len(setups)==63
    timed=[r for r in rows if not r['error']]
    measured={'wall_all_attempts_ms':dist([r['elapsed_ms'] for r in rows]),
        'wall_ms':dist([r['elapsed_ms'] for r in timed]),'cpu_ms':dist([r['cpu_ms'] for r in timed]),
        'query_process_peak_rss_GiB':dist([p['peak_rss_native']/2**30 for p in provs]),
        'setup_process_peak_rss_GiB':dist([p['peak_rss_native']/2**30 for p in setups]),
        'prepare_and_warmup_ms_by_repo':dist([p['preparation_and_warmup_ms'] for p in provs]),
        'static_index_ms_by_repo_shared_between_arms':dist([p['index_ms'] for p in setups]),
        'new_context_source_vectors':sum(p['new_vectors'] for p in prefills),
        'reused_context_source_vectors':sum(p['reused'] for p in prefills),
        'source_prefill_wall_s_total':sum(p['elapsed_ms'] for p in prefills)/1000}
    for k,v in measured.items():
        if isinstance(v,dict):
            for n,x in v.items():assert math.isclose(x,summary['costs'][arm][k][n],abs_tol=1e-9),(arm,k,n)
        else:assert math.isclose(v,summary['costs'][arm][k],abs_tol=1e-9),(arm,k)
print(json.dumps({'verified':True,'current_calls':3753,'historical_calls':2502,
    'all_registered_metrics_and_costs_recomputed':True,'models_loaded':False,'private_reserve':'CLOSED'}))
