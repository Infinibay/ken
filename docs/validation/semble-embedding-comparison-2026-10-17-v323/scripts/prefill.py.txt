"""Source-only cache prefill. Parallel preparation ends before query timing."""
from concurrent.futures import ThreadPoolExecutor
import hashlib,json,time
import numpy as np
from ken.retrieval.context_preparation import ContextVectorCache,full_owner_text,_valid
from ken.retrieval.description import describe_owner
from ken.retrieval.source_views import owner_view_texts

def prefill(catalog,encoder,directory,progress):
 started=time.perf_counter();unique={}
 for key in catalog.owners:
  description=describe_owner(catalog.owners[key])
  for text in [full_owner_text(description),*owner_view_texts(description).values()]:
   digest=hashlib.sha256(text.encode()).hexdigest();assert digest not in unique or unique[digest]==text;unique[digest]=text
 texts=list(unique.values());probe=[texts[i] for i in sorted({0,len(texts)//4,len(texts)//2,3*len(texts)//4,len(texts)-1})]
 def embed(text):return np.asarray(encoder.embed_passages([text])[0],dtype=np.float32)
 serial=[embed(t) for t in probe]
 with ThreadPoolExecutor(max_workers=4) as pool:parallel=list(pool.map(embed,probe))
 assert all(np.array_equal(a,b) for a,b in zip(serial,parallel,strict=True)),'parallel source encoding changed bits'
 with ContextVectorCache(directory,encoder.spec) as cache:
  missing=[t for t in texts if cache.get(t) is None];pending=[]
  with ThreadPoolExecutor(max_workers=4) as pool:
   for number,(text,raw) in enumerate(zip(missing,pool.map(embed,missing),strict=True),1):
    value=_valid((raw/np.linalg.norm(raw)).astype(np.float32));pending.append((text,value))
    if len(pending)==64 or number==len(missing):
     cache.put_many(pending);pending=[];progress({'encoded':number,'missing':len(missing),'total_views':len(texts),'elapsed_s':time.perf_counter()-started,'concurrent_source_calls':4,'query_calls_serial':True})
 return {'total_views':len(texts),'new_vectors':len(missing),'reused':len(texts)-len(missing),'elapsed_ms':(time.perf_counter()-started)*1000,'parity_probes':len(probe),'bitwise_serial_parallel_equal':True,'source_concurrency':4}
