"""Reuse the verified native-pooling research adapters without altering Ken."""
import hashlib, json, time
from pathlib import Path
from dataclasses import dataclass, replace
import numpy as np

def sha(p):
 with Path(p).open("rb") as f:return hashlib.file_digest(f,"sha256").hexdigest()

def resources_for(arm, registrations):
 from ken.retrieval.best_resources import default_resources
 resources=default_resources()
 reg=registrations.get(arm,{})
 if arm=='jina':
  import onnxruntime as ort
  from tokenizers import Tokenizer
  from ken.retrieval.profile import EncoderSpec,ContextualRetrieval
  modelroot=Path(reg['model_root'])
  for n,d in reg['model_artifacts'].items():assert sha(modelroot/n)==d
  @dataclass(frozen=True)
  class Model:
   document_projection:str=resources.context.model.document_projection
   @property
   def spec(self):
    identity={'revision':reg['jina_revision'],'artifacts':reg['model_artifacts'],'projection':self.document_projection,'pooling':'mean-mask-L2','prefix':'','tokens':512,'batch':1,'threads':[2,1],'runtime':{n:__import__('importlib.metadata').metadata.version(n) for n in ['onnxruntime','tokenizers','numpy']}}
    return EncoderSpec('jinaai/jina-embeddings-v2-base-code',hashlib.sha256(json.dumps(identity,sort_keys=True).encode()).hexdigest(),'provided',sequence_length=512,batch_size=1)
  class Encoder:
   def __init__(self):
    self.model=Model();self.spec=self.model.spec
    options=ort.SessionOptions();options.intra_op_num_threads=2;options.inter_op_num_threads=1
    self.session=ort.InferenceSession(str(modelroot/'onnx/model_quantized.onnx'),sess_options=options,providers=['CPUExecutionProvider']);self.tokenizer=Tokenizer.from_file(str(modelroot/'tokenizer.json'));self.tokenizer.no_padding();self.tokenizer.enable_truncation(512);self.calls=0;self.encode_ms=0.
   @property
   def providers(self):return tuple(self.session.get_providers())
   def _encode(self,text):
    started=time.perf_counter();t=self.tokenizer.encode(text);mask=np.array([t.attention_mask],np.int64);hidden=self.session.run(['last_hidden_state'],{'input_ids':np.array([t.ids],np.int64),'attention_mask':mask})[0];v=(hidden*mask[:,:,None]).sum(axis=1)/mask.sum(axis=1,keepdims=True);v=v[0].astype(np.float32);assert v.shape==(768,) and np.isfinite(v).all();self.calls+=1;self.encode_ms+=(time.perf_counter()-started)*1000
    return v
   def embed_query(self,text):return self._encode(text)
   def embed_passages(self,texts):return [self._encode(t) for t in texts]
  encoder=Encoder();view=replace(encoder.model,document_projection=__import__('ken.retrieval.source_views',fromlist=['SOURCE_VIEW_PROJECTION']).SOURCE_VIEW_PROJECTION).spec
  resources=replace(resources,context=encoder,profile=replace(resources.profile,contextual=ContextualRetrieval(encoder.spec,view_model=view)))
  import ken.retrieval.best_search as bs
  import ken.retrieval.best_resources as br
  bs.default_resources=lambda:resources;br.default_resources=lambda:resources
 if arm=='muninn':
  import onnxruntime as ort
  from tokenizers import Tokenizer
  from ken.retrieval.profile import EncoderSpec,ContextualRetrieval
  modelroot=Path(reg['model_root'])
  for n,d in reg['model_artifacts'].items():assert sha(modelroot/n)==d
  @dataclass(frozen=True)
  class Model:
   document_projection:str=resources.context.model.document_projection
   @property
   def spec(self):
    identity={'revision':reg['muninn_revision'],'artifacts':reg['model_artifacts'],'projection':self.document_projection,'pooling':'native-CLS-L2','query_prefix':reg['policy']['query_prefix'],'document_prefix':reg['policy']['document_prefix'],'zero_padding':'384-to768-exact-zeros','tokens':512,'batch':1,'threads':[2,1],'runtime':{n:__import__('importlib.metadata').metadata.version(n) for n in ['onnxruntime','tokenizers','numpy']}}
    return EncoderSpec('brokkai/Muninn-small',hashlib.sha256(json.dumps(identity,sort_keys=True).encode()).hexdigest(),'provided',sequence_length=512,batch_size=1)
  class Encoder:
   def __init__(self):
    self.model=Model();self.spec=self.model.spec
    options=ort.SessionOptions();options.intra_op_num_threads=2;options.inter_op_num_threads=1
    self.session=ort.InferenceSession(str(modelroot/'onnx/model.onnx'),sess_options=options,providers=['CPUExecutionProvider']);self.tokenizer=Tokenizer.from_file(str(modelroot/'tokenizer.json'));self.tokenizer.no_padding();self.tokenizer.enable_truncation(512);self.calls=0;self.encode_ms=0.
   @property
   def providers(self):return tuple(self.session.get_providers())
   def _encode(self,text):
    started=time.perf_counter();t=self.tokenizer.encode(text);mask=np.array([t.attention_mask],np.int64);v=self.session.run(['sentence_embedding'],{'input_ids':np.array([t.ids],np.int64),'attention_mask':mask})[0][0].astype(np.float32);assert v.shape==(384,) and np.isfinite(v).all();v=np.pad(v,(0,384));assert v.shape==(768,) and not v[384:].any();self.calls+=1;self.encode_ms+=(time.perf_counter()-started)*1000

   def embed_query(self,text):return self._encode(reg['policy']['query_prefix']+text)
   def embed_passages(self,texts):return [self._encode(reg['policy']['document_prefix']+t) for t in texts]
  encoder=Encoder();view=replace(encoder.model,document_projection=__import__('ken.retrieval.source_views',fromlist=['SOURCE_VIEW_PROJECTION']).SOURCE_VIEW_PROJECTION).spec
  resources=replace(resources,context=encoder,profile=replace(resources.profile,contextual=ContextualRetrieval(encoder.spec,view_model=view)))
  import ken.retrieval.best_search as bs
  import ken.retrieval.best_resources as br
  bs.default_resources=lambda:resources;br.default_resources=lambda:resources
 return resources
